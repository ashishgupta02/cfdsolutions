# Mesh2D executable

file(GLOB_RECURSE MESH2D_MESHER_SOURCES MESHER/src/*.cpp)
file(GLOB_RECURSE MESH2D_MESHIO_SOURCES MESHIO/src/*.cpp)
file(GLOB_RECURSE MESH2D_OPTIMESH_SOURCES OPTIMESH/src/*.cpp)
file(GLOB_RECURSE MESH2D_WINSLOW_SOURCES WINSLOW/src/*.cpp)
file(GLOB_RECURSE MESH2D_WINSLOWVV_SOURCES WINSLOWVV/src/*.cpp)
file(GLOB_RECURSE MESH2D_LES_SOURCES LESMOOTH/src/*.cpp)
file(GLOB_RECURSE MESH2D_ADAPT_SOURCES ADAPTREFINE/src/*.cpp)

add_executable(mesh2d main.cpp ${MESH2D_MESHER_SOURCES} ${MESH2D_MESHIO_SOURCES} ${MESH2D_OPTIMESH_SOURCES} ${MESH2D_WINSLOW_SOURCES} ${MESH2D_WINSLOWVV_SOURCES} ${MESH2D_LES_SOURCES} ${MESH2D_ADAPT_SOURCES})

target_include_directories(mesh2d PRIVATE
  ${CMAKE_CURRENT_SOURCE_DIR}
  ${CMAKE_CURRENT_SOURCE_DIR}/MESHER/include
  ${CMAKE_CURRENT_SOURCE_DIR}/MESHIO/include
  ${CMAKE_CURRENT_SOURCE_DIR}/OPTIMESH/include
  ${CMAKE_CURRENT_SOURCE_DIR}/WINSLOW/include
  ${CMAKE_CURRENT_SOURCE_DIR}/WINSLOWVV/include
  ${CMAKE_CURRENT_SOURCE_DIR}/LESMOOTH/include
  ${CMAKE_CURRENT_SOURCE_DIR}/ADAPTREFINE/include
  ${CMAKE_SOURCE_DIR}/UTILS/include
  ${CMAKE_SOURCE_DIR}/MATH/include
  ${CMAKE_SOURCE_DIR}
)

target_link_libraries(mesh2d PRIVATE UTILS MATH)

# Apply project-wide compile options
target_link_libraries(mesh2d PRIVATE project_compile_options)

target_compile_features(mesh2d PRIVATE cxx_std_11)
